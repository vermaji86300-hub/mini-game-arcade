    <h2>⚡ Color Reflex</h2>
    <p>Click the matching color circle before time runs out!</p>
    <div id="target" class="color-target">READY</div>
    <div class="options" id="color-options"></div>
    <p>Score: <span id="reflexScore">0</span> | Time Left: <span id="timer">3</span>s</p>

  <script>
    const palette = ["#ef4444", "#3b82f6", "#10b981", "#eab308"];
    const names = ["RED", "BLUE", "GREEN", "YELLOW"];
    let score = 0, timerVal = 3, interval, targetIdx;

    function nextRound() {
      clearInterval(interval);
      timerVal = 3;
      document.getElementById("timer").innerText = timerVal;
      targetIdx = Math.floor(Math.random() * palette.length);
      const targetDiv = document.getElementById("target");
      targetDiv.innerText = names[targetIdx];
      targetDiv.style.color = palette[Math.floor(Math.random() * palette.length)]; // trick color

      interval = setInterval(() => {
        timerVal--;
        document.getElementById("timer").innerText = timerVal;
        if (timerVal <= 0) {
          clearInterval(interval);
          alert("Time Out! Final Score: " + score);
        }
      }, 1000);
    }

    function checkChoice(idx) {
      if (idx === targetIdx) {
        score++;
        document.getElementById("reflexScore").innerText = score;
        nextRound();
      } else {
        clearInterval(interval);
        alert("Wrong Color! Game Over. Score: " + score);
      }
    }

    function startGame() {
      score = 0;
      document.getElementById("reflexScore").innerText = score;
      const opts = document.getElementById("color-options");
      opts.innerHTML = "";
      palette.forEach((col, idx) => {
        let b = document.createElement("div");
        b.className = "color-btn";
        b.style.backgroundColor = col;
        b.onclick = () => checkChoice(idx);
        opts.appendChild(b);
      });
      nextRound();
    }
  </script>
